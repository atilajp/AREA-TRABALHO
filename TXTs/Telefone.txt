# cria lista de sessões parseando saída do qwinsta
$sessions = (qwinsta) | ForEach-Object {
    $line = $_.Trim()
    if ($line -and -not $line.StartsWith("SESSIONNAME") -and -not $line.StartsWith("USERNAME")) {
        $cols = $line -split '\s+'
        # Usuário, NomeSessão, Id, Estado ...
        [PSCustomObject]@{ Username = $cols[0]; SessionName = $cols[1]; Id = $cols[2]; State = $cols[3] }
    }
} | Where-Object { $_ -ne $null }

# identifica id da sessão atual (marcada com '>' em qwinsta)
$currentId = (qwinsta) | ForEach-Object {
    if ($_ -match '^\s*>') { ($_.TrimStart() -split '\s+')[2] }
} | Select-Object -First 1

# desconecta todas exceto a sessão atual
foreach ($s in $sessions) {
    if ($s.Id -and ($s.Id -ne $currentId)) {
        Write-Host "Desconectando ID $($s.Id) - $($s.Username) ..."
        logoff $s.Id /V
    }
}
