Public Function Util_getPrecoCustoProduto(ByVal IDProduto As Integer) As Double

    Dim dblPrecoCustoTemp As Double
    Dim rsProduto As DAO.Recordset
    Dim strSQL As String, strTipoPrecificacao As String
    
    strSQL = "SELECT * FROM [Cadastro de Mercadorias] WHERE [Código da Mercadoria] = " & IDProduto
    
    Set rsProduto = CurrentDb.OpenRecordset(strSQL, dbOpenSnapshot)
    
    With rsProduto
    
        If .EOF Then
            Util_getPrecoCustoProduto = 0
            Exit Function
        End If
            
        strTipoPrecificacao = Util_getCampoEmpresa("TipoPrecificacao")
        
        dblPrecoCustoTemp = (.Fields("preço c") + (.Fields("preço c") * _
        ((IIf(strTipoPrecificacao = "TVA", 0, -.Fields("ICMS")) + .Fields("IPI") + .Fields("Frete") + .Fields("Emb") + .Fields("Financ")) / 100)) + _
        IIf(.Fields("o") = 0, 0, IIf(strTipoPrecificacao = "TVA", (((.Fields("preço c") + (.Fields("preço c") * ((.Fields("ipi")) / 100))) * (1 + (.Fields("o") / 100))) * 0.17) - (.Fields("preço c") * (.Fields("ICMS") / 100)), (.Fields("preço c") * ((.Fields("o")) / 100)))))
        
        Util_getPrecoCustoProduto = Format(dblPrecoCustoTemp, "0.00")
    
        .Close
    
    End With
    
End Function
